```python
NAME_PAIR_RE = re.compile(r'[A-Z][a-z]+(?:[-\']?[A-Z][a-z]+)?\s+[A-Z][a-z]+')
EMAIL_RE = re.compile(r'[\w.+-]+@[\w-]+\.[a-zA-Z]{2,}')

def is_author_block(block: str, index: int, total_blocks: int) -> bool:
    # Limit to early document region
    if index > 6 and index >= 0.12 * total_blocks:
        return False

    # Semicolon heuristic
    if block.count(';') >= 1 and (NAME_PAIR_RE.search(block) or AFFIL_RE.search(block) or EMAIL_RE.search(block)):
        return True

    # Bolded names heuristic
    bold_names = re.findall(r'\*\*([^*]{2,80}?)\*\*', block)
    if len(bold_names) >= 2:

    lines = [l.strip() for l in block.splitlines() if l.strip()]
    found_names = sum(1 for l in lines if NAME_PAIR_RE.search(l))
    if found_names >= 2:
        return True

    # Affiliation keyword + name combination
    if AFFIL_RE.search(block) and NAME_PAIR_RE.search(block):
        return True

    return False
```

```python
MATH_SYMBOLS_RE = re.compile(r'[=^_{}\\]|\\[a-zA-Z]+')

def is_math_block(block: str):
    if not re.search(r'(\$\$|\\\[|\\\(|\$)', block):
        return False
    dollar_regions = re.findall(r'\${1,2}(.+?)\${1,2}', block, flags=re.S)
    for region in dollar_regions:
        tokens = re.findall(r'\S+', region)
        if not tokens:
            continue
        math_tokens = sum(1 for t in tokens if MATH_SYMBOLS_RE.search(t))
        if math_tokens / len(tokens) > 0.25:
            return True
    if r'\begin{' in block and r'\end{' in block:
        return True
    return False
```

### **2️⃣ Update `is_author_block()`**

Replace the function with this refined version:

```python
import re

NAME_PAIR_RE = re.compile(r'[A-Z][a-z]+(?:[-\']?[A-Z][a-z]+)?\s+[A-Z][a-z]+')
EMAIL_RE = re.compile(r'[\w.+-]+@[\w-]+\.[a-zA-Z]{2,}')
AFFIL_RE = re.compile(r'\b(University|Institute|Dept|College|Laboratory|Stanford|MIT|Harvard|Research)\b', re.I)

def is_author_block(block: str, index: int, total_blocks: int) -> bool:
    lines = [l.strip() for l in block.splitlines() if l.strip()]
    text = ' '.join(lines)

    # 🚫 Negative rules
    if '|' in text:
        return False
    if re.match(r'^(\d+\.)|^-|^#', text):
        return False
    if len(lines) > 5 or len(text) > 300:
        return False
    if sum(1 for l in lines if re.match(r'^\d+\.', l)) >= 2:
        return False

    # ✅ Positive signals
    if index < 6 or index < 0.1 * total_blocks:
        # Bolded names or semicolon fields
        if re.findall(r'\*\*[A-Za-z ,.\'-]+\*\*', block):
            return True
        if block.count(';') >= 1 and (NAME_PAIR_RE.search(block) or AFFIL_RE.search(block)):
            return True
        if EMAIL_RE.search(block):
            return True
        # Multiple lines each with capitalized names
        found_names = sum(1 for l in lines if NAME_PAIR_RE.search(l))
        if found_names >= 2:
            return True
    return False
```
